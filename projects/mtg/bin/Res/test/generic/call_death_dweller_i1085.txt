#From the #1085 meta-list: Call of the Death-Dweller (no symptom
#given). Simple path: return one mana-value-2 creature from the
#graveyard to the battlefield.
# https://github.com/WagicProject/wagic/issues/1085
#Note: its persistent global effect uses the `emblem` keyword but is NOT a
#real emblem, so (issue #23) it carries no emblem("...") marker and shows no
#command-zone card - the command zone stays empty here.
[INIT]
FIRSTMAIN
[PLAYER1]
hand:call of the death-dweller
graveyard:grizzly bears
manapool:{2}{B}
[PLAYER2]
[DO]
call of the death-dweller
choice 0
grizzly bears
goto firstmain
goto firstmain
choice 0
goto secondmain
goto secondmain
[ASSERT]
SECONDMAIN
[PLAYER1]
inplay:grizzly bears
graveyard:call of the death-dweller
[PLAYER2]
[END]
