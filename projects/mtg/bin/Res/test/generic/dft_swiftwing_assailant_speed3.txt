#turns of an opponent losing life. Seeded speed is above 0, so Start your engines! does nothing
#and no speed tracker token appears.

[PLAYER1]

[PLAYER2]
life:20
[DO]
[ASSERT]
FIRSTMAIN
[PLAYER1]
inplay:Swiftwing Assailant
pt:Swiftwing Assailant=3/3
life:20
speed:3
[PLAYER2]
life:20
[END]
